\chapter{Spis skrótów i~symboli}

\begin{itemize}
    \item[p.p.] punkt procentowy,
    \item[IoU] Intersection over Union -- miara stopnia pokrycia ramki przewidywanej przez model z~ramką referencyjną,
    \item[TP] True Positive -- prawdziwy pozytyw, poprawnie wykryty obiekt,
    \item[FP] False Positive -- fałszywy pozytyw, błędna detekcja zwrócona przez model,
    \item[FN] False Negative -- fałszywy negatyw, obiekt rzeczywisty niewykryty przez model,
    \item[PR] Precision-Recall -- zależność między precyzją a~czułością,
    \item[AP] Average Precision -- średnia precyzja wyznaczana na podstawie krzywej precision-recall,
    \item[mAP] mean Average Precision -- średnia wartość AP obliczana dla wszystkich klas,
    \item[F1] $F1$-score -- średnia harmoniczna precyzji i~czułości,
    \item[VOC] Visual Object Classes -- zbiórnazwa benchmarku PASCAL VOC wykorzystywanego w~ocenie detekcji obiektów,
    \item[COCO] Common Objects in Context -- nazwa zbióru i~benchmarku Microsoft COCO,
    \item[W3C] World Wide Web Consortium -- organizacja zajmująca się standaryzacją technologii internetowych,
    \item[WCAG] Web Content Accessibility Guidelines -- wytyczne dotyczące dostępności treści internetowych,
    \item[CNN] Convolutional Neural Network -- konwolucyjna sieć neuronowa,
    \item[HOG] Histogram of Oriented Gradients -- histogramy gradientów kierunkowych,
    \item[SIFT] Scale-Invariant Feature Transform -- transformacja cech niezależnych,
    \item[RPN] Region Proposal Network -- sieć generująca propozycje regionów,
    \item[YOLO] You Only Look Once -- rodzina modeli detekcji jednoetapowej obiektów w~czasie rzeczywistym,
    \item[DETR] DEtection TRansformer -- model detekcji obiektów wykorzystujący architekturę transformera,
    \item[NMS] Non-Maximum Suppression -- algorytm usuwania nakładających się ramek detekcji,
    \item[RT] Real-Time -- czas rzeczywisty,
    \item[SSD] Single Shot MultiBox Detector -- model detekcji jednoetapowej obiektów w~czasie rzeczywistym,
    \item[FPS] Frames Per Second -- liczba kadrów na sekundę,
    \item[DFL] Distribution Focal Loss -- funkcja straty wykorzystywana w~modelach detekcji obiektów,
    \item[RGB] Red-Green-Blue -- przestrzeń kolorów reprezentowana przez trzy kanały: czerwony, zielony i~niebieski,
    \item[ONZ] Organizacja Narodów Zjednoczonych -- organizacja międzynarodowa zrzeszająca państwa świata,
    \item[GPS] Global Positioning System -- globalny system pozycjonowania satelitarnego,
    \item[3D] Three-Dimensional -- trójwymiarowy sposób reprezentacji obiektów lub przestrzeni,
    \item[YCB] Yale--Carnegie Mellon University--Berkeley -- zbiór obiektów i~modeli wykorzystywany w~badaniach nad manipulacją robotyczną,
    \item[SVM] Support Vector Machine -- maszyna wektorów nośnych wykorzystywana do klasyfikacji danych,
    \item[R-CNN] Region-based Convolutional Neural Network -- konwolucyjna sieć neuronowa analizująca proponowane regiony obrazu,
    \item[ECCV] European Conference on Computer Vision -- europejska konferencja naukowa dotycząca widzenia komputerowego,
    \item[AI] Artificial Intelligence -- sztuczna inteligencja,
    \item[$mAP50$] mean Average Precision at IoU 0.50 -- średnia wartość AP wyznaczana przy progu IoU równym 0,5,
    \item[$mAP50\text{-}95$] mean Average Precision at IoU 0.50--0.95 -- średnia wartość AP obliczana dla progów IoU od 0,5 do 0,95,
    \item[FPN] Feature Pyramid Network -- sieć piramidy cech umożliwiająca analizę obiektów w~wielu skalach,
    \item[CSP] Cross Stage Partial -- mechanizm częściowego podziału i~ponownego łączenia map cech,
    \item[C3] CSP Bottleneck with 3 Convolutions -- blok sieci wykorzystujący mechanizm CSP i~trzy operacje konwolucji,
    \item[PAN] Path Aggregation Network -- sieć agregacji ścieżek służąca do wieloskalowego łączenia cech,
    \item[SPPF] Spatial Pyramid Pooling -- Fast -- szybki moduł przestrzennego grupowania piramidowego,
    \item[C2f] CSP Bottleneck with two convolutions and feature fusion -- blok CSP wykorzystujący dwie operacje konwolucji i~fuzję cech,
    \item[C3k2] C3k2 block -- wariant bloku CSP wykorzystujący moduły C3k lub bloki typu bottleneck,
    \item[C2PSA] Cross-Stage Partial Spatial Attention -- blok CSP wykorzystujący mechanizm uwagi przestrzennej,
    \item[STAL] Small-Target-Aware Label Assignment -- mechanizm przypisywania etykiet uwzględniający małe obiekty,
    \item[MuSGD] Muon-inspired Stochastic Gradient Descent -- hybrydowy optymalizator łączący SGD z~rozwiązaniami inspirowanymi optymalizatorem Muon,
    \item[GFLOPs] Giga Floating-Point Operations -- liczba miliardów operacji zmiennoprzecinkowych,
    \item[CMOS] Complementary Metal-Oxide-Semiconductor -- technologia wykonywania światłoczułych sensorów obrazu,
    \item[CCD] Charge-Coupled Device -- rodzaj światłoczułego sensora obrazu wykorzystującego elementy ze sprzężeniem ładunkowym,
    \item[JPEG] Joint Photographic Experts Group -- stratny format kompresji i~zapisu obrazów,
    \item[HEIF] High Efficiency Image File Format -- wysokowydajny format zapisu obrazów,
    \item[LiDAR] Light Detection and Ranging -- technologia pomiaru odległości za pomocą światła laserowego,
    \item[RGB-D] Red-Green-Blue and Depth -- reprezentacja danych zawierająca obraz kolorowy i~informację o~głębi,
    \item[SUN] Scene Understanding -- nazwa zbioru danych przeznaczonego do badań nad rozumieniem scen,
    \item[KITTI] Karlsruhe Institute of Technology and Toyota Technological Institute -- zbiór danych i~zestaw benchmarków wykorzystywany w~badaniach nad autonomiczną jazdą i~widzeniem komputerowym,
    \item[YOLO-OD] You Only Look Once--Obstacle Detection -- model detekcji przeszkód przeznaczony do wspomagania nawigacji osób z~niepełnosprawnością wzroku,
    \item[YOLO-POINT] YOLO-POINT -- model łączący detekcję YOLO z~informacjami pochodzącymi z~obrazu i~chmury punktów,
    \item[DTC-YOLO] Depth-Texture Coupling Mechanism YOLO -- multimodalny model detekcji wykorzystujący połączenie informacji o~głębi i~teksturze,
    \item[ADF\textsuperscript{3}-Net] Adaptive Dimension-aware Focused Fusion Network -- sieć adaptacyjnej fuzji cech pochodzących z~różnych wymiarów i~poziomów,
    \item[VI3DR] Visually Impaired 3D Recognition -- system wspomagający rozpoznawanie obiektów w~3D dla osób z~niepełnosprawnością wzroku,
    \item[NSD] Network Service Discovery -- mechanizm wykrywania usług sieciowych w~sieci lokalnej,
    \item[CPU] Central Processing Unit -- procesor centralny,
    \item[GPU] Graphics Processing Unit -- procesor graficzny,
    \item[MPS] Metal Performance Shaders -- framework do obliczeń równoległych na procesorach graficznych Apple,
    \item[Qt] Qt -- framework do tworzenia aplikacji wieloplatformowych,
    \item[S3] Amazon Simple Storage Service -- usługa przechowywania danych w~chmurze firmy Amazon,
    \item[FLOPs] Floating-Point Operations -- liczba operacji zmiennoprzecinkowych,

\end{itemize}
